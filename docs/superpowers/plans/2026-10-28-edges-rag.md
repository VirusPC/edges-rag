# Edges RAG Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a FastAPI knowledge Q&A service that indexes the composed VirusPC/edges tree into local Chroma, keeps it fresh with watchdog, and answers via Ollama (Tailscale) with Bearer auth and citations.

**Architecture:** On startup, walk `INDEX_ROOT` with an extension allowlist and directory denylist, embed documents through Ollama, and persist vectors in Chroma. A watchdog observer applies incremental upserts/deletes by source path. `POST /ask` authenticates with `RAG_API_TOKEN`, retrieves top-k, generates an answer with the Ollama chat model, and returns `{ answer, citations }`.

**Tech Stack:** Python 3.11+, LlamaIndex, FastAPI, uvicorn, chromadb, watchdog, pydantic-settings, pytest, httpx (via LlamaIndex Ollama integrations).

**Spec:** docs/superpowers/specs/2026-10-28-edges-rag-design.md

## Global Constraints

- Repo target: VirusPC/edges-rag, public; no secrets in tree.
- Knowledge source: configurable `INDEX_ROOT` (composed edges tree on minigtr); not personal brain dumps.
- Index file types (default allowlist): `.md`, `.txt`, `.py`, `.ts`, `.tsx`, `.js`, `.jsx`, `.json`, `.yaml`, `.yml`, `.toml`, `.sh`, `.rs`, `.go`.
- Exclude dirs (default): `node_modules`, `.git`, `__pycache__`, `.venv`, `dist`, `build`; also skip binary / undecodable files.
- Index lifecycle: full scan on startup + watchdog realtime incremental add/update/delete.
- Vector store: local Chroma, persist path configurable (`CHROMA_PERSIST_DIR`).
- LLM + embeddings: Ollama via `OLLAMA_BASE_URL` (Tailscale); models via `OLLAMA_EMBED_MODEL` / `OLLAMA_CHAT_MODEL`.
- API: FastAPI `POST /ask` → `{ answer, citations: [{ path, snippet, score }] }`.
- Auth: Bearer token from `RAG_API_TOKEN`; missing/wrong → 401.
- No MySQL. No OpenAI-compatible surface in v1.
- Python package layout under `src/edges_rag/`.
- TDD for every task: failing test → run fail → impl → run pass → commit.

```python
from __future__ import annotations

from functools import lru_cache
from pathlib import Path

from pydantic import Field, field_validator
from pydantic_settings import BaseSettings, SettingsConfigDict

_DEFAULT_EXTS = (
    ".md,.txt,.py,.ts,.tsx,.js,.jsx,.json,.yaml,.yml,.toml,.sh,.rs,.go"
)
_DEFAULT_EXCLUDE = "node_modules,.git,__pycache__,.venv,dist,build"


def parse_extensions(raw: str) -> list[str]:
    out: list[str] = []
    for part in raw.split(","):
        p = part.strip().lower()
        if not p:
            continue
        if not p.startswith("."):
            p = f".{p}"
        out.append(p)
    return out


def parse_exclude_dirs(raw: str) -> set[str]:
    return {p.strip() for p in raw.split(",") if p.strip()}


class Settings(BaseSettings):
    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        extra="ignore",
    )

    index_root: Path = Field(alias="INDEX_ROOT")
    edges_root: Path | None = Field(default=None, alias="EDGES_ROOT")
    chroma_persist_dir: Path = Field(alias="CHROMA_PERSIST_DIR")
    ollama_base_url: str = Field(alias="OLLAMA_BASE_URL")
    ollama_embed_model: str = Field(alias="OLLAMA_EMBED_MODEL")
    ollama_chat_model: str = Field(alias="OLLAMA_CHAT_MODEL")
    rag_api_token: str = Field(alias="RAG_API_TOKEN")
    rag_top_k: int = Field(default=5, alias="RAG_TOP_K")
    rag_host: str = Field(default="0.0.0.0", alias="RAG_HOST")
    rag_port: int = Field(default=8000, alias="RAG_PORT")
    index_allow_extensions_raw: str = Field(
        default=_DEFAULT_EXTS, alias="INDEX_ALLOW_EXTENSIONS"
    )
    index_exclude_dirs_raw: str = Field(
        default=_DEFAULT_EXCLUDE, alias="INDEX_EXCLUDE_DIRS"
    )

    @property
    def index_allow_extensions(self) -> list[str]:
        return parse_extensions(self.index_allow_extensions_raw)

    @property
    def index_exclude_dirs(self) -> set[str]:
        return parse_exclude_dirs(self.index_exclude_dirs_raw)

    @field_validator("index_root", "chroma_persist_dir", mode="before")
    @classmethod
    def _pathify(cls, v: object) -> object:
        if isinstance(v, str):
            return Path(v)
        return v


@lru_cache
def get_settings() -> Settings:
    return Settings()  # type: ignore[call-arg]
```

```python
from __future__ import annotations

from pathlib import Path

from llama_index.core import Document


def is_under_root(path: Path, root: Path) -> bool:
    try:
        path.resolve().relative_to(root.resolve())
        return True
    except ValueError:
        return False


def should_index_file(
    path: Path,
    root: Path,
    allow_exts: list[str],
    exclude_dirs: set[str],
) -> bool:
    if not path.is_file():
        return False
    if not is_under_root(path, root):
        return False
    rel = path.resolve().relative_to(root.resolve())
    if any(part in exclude_dirs for part in rel.parts[:-1]):
        return False
    if path.suffix.lower() not in allow_exts:
        return False
    return True


def load_documents(
    root: Path,
    allow_exts: list[str],
    exclude_dirs: set[str],
) -> list[Document]:
    root = root.resolve()
    docs: list[Document] = []
    if not root.is_dir():
        return docs
    for path in root.rglob("*"):
        if not should_index_file(path, root, allow_exts, exclude_dirs):
            continue
        try:
            text = path.read_text(encoding="utf-8")
        except (UnicodeDecodeError, OSError):
            continue
        rel = path.resolve().relative_to(root).as_posix()
        docs.append(Document(text=text, metadata={"path": rel}))
    return docs
```

```python
from llama_index.embeddings.ollama import OllamaEmbedding
from llama_index.vector_stores.chroma import ChromaVectorStore

from edges_rag.config import Settings
from edges_rag.loader import load_documents, should_index_file


class IndexStore:
    COLLECTION = "edges_rag"

    def __init__(
        self,
        settings: Settings,
        embed_model: BaseEmbedding | None = None,
    ) -> None:
        self.settings = settings
        self.settings.chroma_persist_dir.mkdir(parents=True, exist_ok=True)
        self._embed = embed_model or OllamaEmbedding(
            model_name=settings.ollama_embed_model,
            base_url=settings.ollama_base_url,
        )
        self._client = chromadb.PersistentClient(
            path=str(settings.chroma_persist_dir)
        )
        self._collection = self._client.get_or_create_collection(self.COLLECTION)
        self._vector_store = ChromaVectorStore(chroma_collection=self._collection)
        self._storage_context = StorageContext.from_defaults(
            vector_store=self._vector_store
        )
        self._index = VectorStoreIndex.from_vector_store(
            self._vector_store,
            embed_model=self._embed,
            storage_context=self._storage_context,
        )

    def get_index(self) -> VectorStoreIndex:
        return self._index

    def build_full(self) -> int:
        # Reset collection for a clean full scan
        self._client.delete_collection(self.COLLECTION)
        self._collection = self._client.get_or_create_collection(self.COLLECTION)
        self._vector_store = ChromaVectorStore(chroma_collection=self._collection)
        self._storage_context = StorageContext.from_defaults(
            vector_store=self._vector_store
        )
        docs = load_documents(
            self.settings.index_root,
            self.settings.index_allow_extensions,
            self.settings.index_exclude_dirs,
        )
        for doc in docs:
            doc.doc_id = str(doc.metadata["path"])
        self._index = VectorStoreIndex.from_documents(
            docs,
            storage_context=self._storage_context,
            embed_model=self._embed,
        )
        return len(docs)

    def _rel_path(self, path: Path) -> str:
        return path.resolve().relative_to(
            self.settings.index_root.resolve()
        ).as_posix()

    def upsert_file(self, path: Path) -> None:
        root = self.settings.index_root
        if not should_index_file(
            path,
            root,
            self.settings.index_allow_extensions,
            self.settings.index_exclude_dirs,
        ):
            return
        try:
            text = path.read_text(encoding="utf-8")
        except (UnicodeDecodeError, OSError):
            return
        rel = self._rel_path(path)
        self.delete_file(path)
        doc = Document(text=text, metadata={"path": rel})
        doc.doc_id = rel
        self._index.insert(doc)

    def delete_file(self, path: Path) -> None:
        try:
            rel = self._rel_path(path)
        except ValueError:
            # If path already relative string usage from watcher delete of missing file:
            rel = Path(path).as_posix()
        try:
            self._index.delete_ref_doc(rel, delete_from_docstore=True)
        except Exception:
            # Also try chroma delete by where filter on metadata path
            try:
                self._collection.delete(where={"path": rel})
            except Exception:
                pass

    def has_path(self, rel_path: str) -> bool:
        try:
            got = self._collection.get(where={"path": rel_path})
            ids = got.get("ids") or []
            return len(ids) > 0
        except Exception:
            return False
```

```python
from __future__ import annotations

from pathlib import Path

from watchdog.events import FileSystemEvent, FileSystemEventHandler
from watchdog.observers import Observer

from edges_rag.index_store import IndexStore


class IndexEventHandler(FileSystemEventHandler):
    def __init__(self, store: IndexStore) -> None:
        super().__init__()
        self.store = store

    def on_created(self, event: FileSystemEvent) -> None:
        if event.is_directory:
            return
        self.store.upsert_file(Path(str(event.src_path)))

    def on_modified(self, event: FileSystemEvent) -> None:
        if event.is_directory:
            return
        self.store.upsert_file(Path(str(event.src_path)))

    def on_deleted(self, event: FileSystemEvent) -> None:
        if event.is_directory:
            return
        self.store.delete_file(Path(str(event.src_path)))

    def on_moved(self, event: FileSystemEvent) -> None:
        if event.is_directory:
            return
        self.store.delete_file(Path(str(event.src_path)))
        dest = getattr(event, "dest_path", None)
        if dest:
            self.store.upsert_file(Path(str(dest)))


def start_watcher(store: IndexStore) -> Observer:
    handler = IndexEventHandler(store)
    observer = Observer()
    observer.schedule(
        handler,
        str(store.settings.index_root.resolve()),
        recursive=True,
    )
    observer.start()
    return observer
```

```python
from __future__ import annotations

from llama_index.core.llms import LLM
from llama_index.llms.ollama import Ollama

from edges_rag.config import Settings
from edges_rag.index_store import IndexStore
from edges_rag.schemas import AskResponse, Citation

_SNIPPET_MAX = 240


def _snippet(text: str) -> str:
    cleaned = " ".join(text.split())
    if len(cleaned) <= _SNIPPET_MAX:
        return cleaned
    return cleaned[: _SNIPPET_MAX - 1] + "…"


class QueryService:
    def __init__(
        self,
        store: IndexStore,
        settings: Settings,
        llm: LLM | None = None,
    ) -> None:
        self.store = store
        self.settings = settings
        self.llm = llm or Ollama(
            model=settings.ollama_chat_model,
            base_url=settings.ollama_base_url,
            request_timeout=120.0,
        )

    def ask(self, question: str) -> AskResponse:
        retriever = self.store.get_index().as_retriever(
            similarity_top_k=self.settings.rag_top_k
        )
        nodes = retriever.retrieve(question)
        citations: list[Citation] = []
        context_parts: list[str] = []
        for node in nodes:
            path = str(node.node.metadata.get("path", ""))
            text = node.node.get_content()
            citations.append(
                Citation(
                    path=path,
                    snippet=_snippet(text),
                    score=float(node.score) if node.score is not None else None,
                )
            )
            context_parts.append(f"[{path}]\n{text}")
        context = "\n\n".join(context_parts) if context_parts else "(no context)"
        prompt = (
            "You are the Edges framework knowledge assistant. "
            "Answer using only the provided context. "
            "If the context is insufficient, say you do not know.\n\n"
            f"Context:\n{context}\n\n"
            f"Question: {question}\n\n"
            "Answer:"
        )
        completion = self.llm.complete(prompt)
        answer = str(completion.text).strip()
        return AskResponse(answer=answer, citations=citations)
```

```python
from __future__ import annotations

import hmac

from fastapi import HTTPException, status
from fastapi.security import HTTPAuthorizationCredentials


def verify_bearer(
    credentials: HTTPAuthorizationCredentials | None,
    expected: str,
) -> None:
    if credentials is None or credentials.scheme.lower() != "bearer":
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Unauthorized",
            headers={"WWW-Authenticate": "Bearer"},
        )
    if not hmac.compare_digest(credentials.credentials, expected):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Unauthorized",
            headers={"WWW-Authenticate": "Bearer"},
        )
```

```python
from __future__ import annotations

from contextlib import asynccontextmanager
from typing import AsyncIterator

from fastapi import Depends, FastAPI
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer

from edges_rag.auth import verify_bearer
from edges_rag.config import Settings, get_settings
from edges_rag.index_store import IndexStore
from edges_rag.query_engine import QueryService
from edges_rag.schemas import AskRequest, AskResponse
from edges_rag.watcher import start_watcher

_bearer_scheme = HTTPBearer(auto_error=False)


def create_app(
    settings: Settings | None = None,
    query_service: QueryService | None = None,
    skip_lifespan_index: bool = False,
) -> FastAPI:
    settings = settings or get_settings()
    state: dict = {
        "settings": settings,
        "query_service": query_service,
        "observer": None,
        "store": None,
    }

    @asynccontextmanager
    async def lifespan(app: FastAPI) -> AsyncIterator[None]:
        if not skip_lifespan_index and state["query_service"] is None:
            store = IndexStore(settings)
            store.build_full()
            state["store"] = store
            state["query_service"] = QueryService(store, settings)
            state["observer"] = start_watcher(store)
        yield
        observer = state.get("observer")
        if observer is not None:
            observer.stop()
            observer.join(timeout=5)

    app = FastAPI(title="edges-rag", lifespan=lifespan)

    async def require_auth(
        credentials: HTTPAuthorizationCredentials | None = Depends(_bearer_scheme),
    ) -> None:
        verify_bearer(credentials, state["settings"].rag_api_token)

    @app.post("/ask", response_model=AskResponse)
    async def ask(
        body: AskRequest,
        _: None = Depends(require_auth),
    ) -> AskResponse:
        svc: QueryService = state["query_service"]
        return svc.ask(body.question)

    app.state.settings_ref = state  # type: ignore[attr-defined]
    return app


# Import-safe default for tooling/tests; real deploys use `python -m edges_rag`
# which calls create_app(skip_lifespan_index=False).
app = create_app(skip_lifespan_index=True)
```
